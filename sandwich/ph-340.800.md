class PentadPDF(FPDF):
    def __init__(self, logo_path, qr_path, font_regular, font_bold):
        super().__init__()
        self.logo_path = logo_path
        self.qr_path = qr_path
        self.font_bold_available = os.path.exists(font_bold)
        self.set_auto_page_break(auto=True, margin=20)
        self.set_margins(left=20, top=30, right=20)
        self.add_font("DejaVu", "", font_regular, uni=True)
        if self.font_bold_available:
            self.add_font("DejaVu", "B", font_bold, uni=True)
        self.set_font("DejaVu", "", 11)
        self.add_page()

    def header(self):
        if os.path.exists(self.logo_path):
            self.image(self.logo_path, x=10, y=10, w=20)
        self.set_font("DejaVu", "B" if self.font_bold_available else "", 16)
        self.cell(0, 10, "Ukubona LLC", ln=True, align="C")
        self.set_font("DejaVu", "", 11)
        self.cell(0, 10, "13675 Bent Tree Cir 201, Centreville, VA 20121", ln=True, align="C")
        self.cell(0, 10, "ukubona.llc@gmail.com | (240) 281-3154", ln=True, align="C")
        if os.path.exists(self.qr_path):
            self.image(self.qr_path, x=180, y=10, w=20)
        self.ln(2)
        self.set_draw_color(200, 200, 200)
        self.set_line_width(0.3)
        self.line(10, self.get_y(), 200, self.get_y())
        self.ln(5)

    def footer(self):
        self.set_y(-20)
        self.set_font("DejaVu", "", 8)
        self.set_text_color(130, 130, 130)
        self.cell(0, 5, "Ukubona LLC reserves all rights. https://ukubona-llc.github.io", ln=True, align="C")
        self.set_y(-10)
        self.set_font("DejaVu", "", 9)
        self.cell(0, 10, f"Page {self.page_no()} - Ukubona LLC (c) 2025", align="C")

    def add_modal_content(self, modal):
        self.set_font("DejaVu", "B" if self.font_bold_available else "", 18)
        self.cell(0, 10, f"{modal['icon']}  {modal['title']}", ln=True)
        self.ln(5)
        self.set_font("DejaVu", "", 12)

                    self.ln(2)

def generate_pentad_pdf(modal_id):

### Example Flask app (`flask_api.py`):

```python
import os
from flask import Flask, jsonify, send_file, abort

app = Flask(__name__)

# Directory where PDFs are stored (adjust as needed)
PDF_DIR = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", "pdfs"))

# Sample modal data matching frontend expectations
MODALS = [
    {
        "id": "bank_letter",
        "title": "Bank Validation Letter",
        "icon": "🏦",
        "filename": "ukubona_bank_validation_letter.pdf"
    },
    {
        "id": "quarterly_report",
        "title": "Quarterly Strategic Report",
        "icon": "📅",
        "filename": "quarterly_strategic_report.pdf"
    },
    {
        "id": "agent_simulation",
        "title": "Agent Simulation Summary",
        "icon": "🤖",
        "filename": "agent_simulation_summary.pdf"
    }
]

@app.route('/api/modals', methods=['GET'])
def get_modals():
    # Return list of modals without filename for security
    modals_list = [{"id": m["id"], "title": m["title"], "icon": m["icon"]} for m in MODALS]
    return jsonify(modals_list)

@app.route('/api/pdf/<modal_id>', methods=['GET'])
def get_pdf(modal_id):
    # Find modal by id
    modal = next((m for m in MODALS if m["id"] == modal_id), None)
    if modal is None:
        abort(404, description="Report not found")

    pdf_path = os.path.join(PDF_DIR, modal["filename"])

    if not os.path.isfile(pdf_path):
        abort(404, description="PDF file not found")

    # Serve PDF with correct mimetype
    return send_file(pdf_path, mimetype='application/pdf', as_attachment=False,
                     download_name=modal["filename"])

if __name__ == "__main__":
    # Optional: debug mode ON for dev
    app.run(host="0.0.0.0", port=5000, debug=True)
```

---

### What you need to do:

1. Place your PDF files in the `../pdfs` directory relative to `flask_api.py`.
2. Add or update `MODALS` list with your actual PDFs and metadata.
3. Run the Flask app (`python flask_api.py`).
4. Frontend `/api/modals` will list your reports, `/api/pdf/<id>` serves the file.

---

### Notes:

* `send_file` with `as_attachment=False` opens PDFs in browser tabs. Change to `True` if you want forced download.
* Adjust `PDF_DIR` if your PDFs live elsewhere.
* You can expand `MODALS` dynamically or load from a config file or DB if needed later.
* This minimal API is production-ready with basic error handling.

  if (!visible) return null;

    data = request.get_json()
