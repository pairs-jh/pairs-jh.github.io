# Backend `auth.py` snippet with token auth & user store (simple):

```python
from flask import Blueprint, request, jsonify, g
from functools import wraps
import hashlib
import secrets
import time

auth_blueprint = Blueprint('auth', __name__)

# Simple in-memory user store (replace with DB in prod)
users = {
    "student1": {
        "password_hash": hashlib.sha256(b"supersecret").hexdigest(),
        "paid": True
    },
    "student2": {
        "password_hash": hashlib.sha256(b"anothersecret").hexdigest(),
        "paid": False
    }
}

tokens = {}

def generate_token(username):
    token = secrets.token_urlsafe(32)
    tokens[token] = {"username": username, "expiry": time.time() + 3600 * 24}  # 24h expiry
    return token

def token_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        token = request.headers.get('Authorization', '').replace('Bearer ', '')
        token_data = tokens.get(token)
        if not token_data or token_data['expiry'] < time.time():
            return jsonify({'error': 'Unauthorized'}), 401
        g.current_user = token_data['username']
        return f(*args, **kwargs)
    return decorated

def login_required(f):
    return token_required(f)

@auth_blueprint.route('/login', methods=['POST'])
def login():
    data = request.json
    username = data.get('username')
    password = data.get('password')
    user = users.get(username)
    if not user:
        return jsonify({'error': 'User not found'}), 404

    if hashlib.sha256(password.encode()).hexdigest() != user['password_hash']:
        return jsonify({'error': 'Invalid credentials'}), 401
    if not user['paid']:
        return jsonify({'error': 'Payment required'}), 402

    token = generate_token(username)
    return jsonify({'token': token})
```

```jsx
import React, { useEffect, useState } from 'react'

function parseCSV(text) {
  const lines = text.trim().split('\n')
  const headers = lines[0].split(',')
  return lines.slice(1).map(line => {
    const values = line.split(',')
    const obj = {}
    headers.forEach((h, i) => {
      obj[h.trim()] = values[i].trim()
    })
    return obj
  })
}

export default function Syllabus({ token }) {
  const [syllabus, setSyllabus] = useState([])

  useEffect(() => {
    async function load() {
      try {
        const resp = await fetch('/api/syllabus', {
          headers: { Authorization: `Bearer ${token}` },
        })
        if (!resp.ok) throw new Error('Failed to load syllabus')
        const text = await resp.text()
        setSyllabus(parseCSV(text))
      } catch {
        setSyllabus([])
      }
    }
    load()
  }, [token])

  if (syllabus.length === 0) return <p>No syllabus data available.</p>

  return (
    <section style={{ padding: '1rem' }}>
      <h3>Course Syllabus</h3>
      <table border="1" cellPadding="6" cellSpacing="0" style={{ width: '100%' }}>
        <thead>
          <tr>
            {Object.keys(syllabus[0]).map(h => (
              <th key={h}>{h}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {syllabus.map((row, i) => (
            <tr key={i}>
              {Object.values(row).map((v, j) => (
                <td key={j}>{v}</td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </section>
  )
}
```
